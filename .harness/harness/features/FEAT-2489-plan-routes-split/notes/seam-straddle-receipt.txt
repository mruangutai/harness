def extract(rev, root):
    os.mkdir(root)
    archive = subprocess.Popen(["git", "archive", rev], cwd=CHECKOUT, stdout=subprocess.PIPE)
    subprocess.run(["tar", "-x", "-C", root], stdin=archive.stdout, check=True)
    archive.stdout.close()
    assert archive.wait() == 0


def defining_file(root, name):
    """The one bin/ top-level .py among the three split files that defines NAME at top level."""
    hits = []
    for base in ("check-plan-routes.py", "canonical_reader_audit.py", "structure_audit.py"):
        path = os.path.join(root, BIN, base)
        if not os.path.isfile(path):
            continue
        tree = ast.parse(open(path, encoding="utf-8").read())
        hits += [(path, node) for node in tree.body
                 if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name == name]
    assert len(hits) == 1, (name, hits)
    return hits[0]


def plant(root, name, lines):
    """Insert LINES as the first statement after NAME's docstring, at the body's indent."""
    path, node = defining_file(root, name)
    body = node.body
    first = body[1] if (len(body) > 1 and isinstance(body[0], ast.Expr)
                        and isinstance(getattr(body[0], "value", None), ast.Constant)
                        and isinstance(body[0].value.value, str)) else body[0]
    source = open(path, encoding="utf-8").read().split("\n")
    indent = " " * first.col_offset
    at = first.lineno - 1
    source[at:at] = [indent + line for line in lines]
    open(path, "w", encoding="utf-8").write("\n".join(source))
    return f"{os.path.relpath(path, root)}::{name} (inserted before line {first.lineno})"


def drop_row(root, row_id):
    path = os.path.join(root, INVENTORY)
    document = json.load(open(path))
    before = len(document["rows"])
    document["rows"] = [row for row in document["rows"] if row.get("id") != row_id]
    assert len(document["rows"]) == before - 1, row_id
    open(path, "w").write(json.dumps(document, indent=2) + "\n")


def run(runner, scanned, argv):
    env = {k: v for k, v in os.environ.items() if k not in SCRUB}
    env.update(PYTHONDONTWRITEBYTECODE="1", HARNESS_PROJECT_DIR=scanned)
    result = subprocess.run([sys.executable, os.path.join(runner, ENTRY), *argv], cwd=scanned,
                            env=env, capture_output=True, timeout=120)
    return result.returncode, result.stdout.decode(), result.stderr.decode()


def record(title, side, runner, scanned, argv):
    code, out, err = run(runner, scanned, argv)
    print(f"----- {title} {side}: runner={runner}/{ENTRY} HARNESS_PROJECT_DIR={scanned} argv={argv}")
    print("--- stdout (whole)")
    print(out, end="" if out.endswith("\n") or not out else "\n[no trailing newline]\n")
    print("--- stderr (whole)")
    print(err, end="" if err.endswith("\n") or not err else "\n[no trailing newline]\n")
    print(f"exit={code}")
    return code, out, err

        record("(r)", side, trees["r", side], trees["r", side], ["--canonical-reader-audit"])
    print()
    print("===== controls: pristine trees (no perturbation), both modes")
    for argv in (["--consolidation-audit"], ["--canonical-reader-audit"]):
        for side in revs:
            code, out, err = run(trees["pristine", side], trees["pristine", side], argv)
            last = out.rstrip("\n").rsplit("\n", 1)[-1]
            print(f"CONTROL {side} {argv[0]} exit={code} last_stdout={last!r} stderr_bytes={len(err)}")
    shutil.rmtree(tmp)
