Run the following measurement script to collect token count estimates for all harness files. The approximation method is chars/4 (no tiktoken available on this machine — see 04-RESEARCH.md for rationale).

```bash
python3 - << 'EOF'
import os
import glob

HARNESS = os.path.expanduser("~/GitHub/harness")

def measure(path, threshold=None, label=None):
    try:
        with open(path) as f:
            content = f.read()
    except Exception as e:
        return {"path": path, "chars": 0, "tokens": 0, "status": f"ERROR: {e}"}
    chars = len(content)
    tokens = chars // 4
    name = label or os.path.basename(path)
    status = "N/A"
    if threshold is not None:
        status = "PASS" if tokens < threshold else "FAIL"
    return {"name": name, "path": path, "chars": chars, "tokens": tokens, "threshold": threshold, "status": status}

measurements = []

# CLAUDE.md — full file (threshold: 1000 tokens per VAL-02 / INFRA-01)
measurements.append(measure(f"{HARNESS}/CLAUDE.md", threshold=1000, label="CLAUDE.md (full)"))

# CLAUDE.md — harness section only (between GSD:harness-start and GSD:harness-end)
with open(f"{HARNESS}/CLAUDE.md") as f:
    content = f.read()
start = content.find("<!-- GSD:harness-start -->")
end = content.find("<!-- GSD:harness-end -->") + len("<!-- GSD:harness-end -->")
harness_section = content[start:end]
measurements.append({
    "name": "CLAUDE.md (harness section only)",
    "path": "CLAUDE.md",
    "chars": len(harness_section),
    "tokens": len(harness_section) // 4,
    "threshold": 200,
    "status": "PASS" if len(harness_section) // 4 < 200 else "FAIL"
})

# Skill files
skill_files = [
    (f"{HARNESS}/.claude/skills/harness/SKILL.md", 500),
    (f"{HARNESS}/.claude/skills/harness/tdd/SKILL.md", 500),
    (f"{HARNESS}/.claude/skills/harness/rules/SKILL.md", 500),
    (f"{HARNESS}/.claude/skills/harness/rules/tdd-enforcement.md", 1000),
    (f"{HARNESS}/.claude/skills/harness/rules/spec-driven.md", 1000),
    (f"{HARNESS}/.claude/skills/harness/rules/systematic-debugging.md", 1000),
    (f"{HARNESS}/.claude/skills/harness/rules/code-review.md", 1000),
    (f"{HARNESS}/.claude/skills/harness/rules/verification-rules.md", 1000),
]
for path, threshold in skill_files:
    measurements.append(measure(path, threshold=threshold))

# Agent files
agent_files = [
    (f"{HARNESS}/.claude/agents/harness-eng-reviewer.md", 2000),
    (f"{HARNESS}/.claude/agents/harness-ceo-reviewer.md", 2000),
    (f"{HARNESS}/.claude/agents/harness-qa-reviewer.md", 2000),
    (f"{HARNESS}/.claude/agents/harness-security-reviewer.md", 2000),
    (f"{HARNESS}/.claude/agents/harness-code-reviewer.md", 500),
]
for path, threshold in agent_files:
    measurements.append(measure(path, threshold=threshold))

print("| File | Chars | ~Tokens | Threshold | Status |")
print("|------|-------|---------|-----------|--------|")
for m in measurements:
    threshold_str = f"<{m['threshold']}" if m.get('threshold') else "N/A"
    print(f"| {m['name']} | {m['chars']} | {m['tokens']} | {threshold_str} | {m['status']} |")

fail_count = sum(1 for m in measurements if m.get('status') == 'FAIL')
pass_count = sum(1 for m in measurements if m.get('status') == 'PASS')
print(f"\nTotal: {pass_count} PASS, {fail_count} FAIL")
EOF
```
